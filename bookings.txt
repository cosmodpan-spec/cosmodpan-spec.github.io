<script>
  const loginBtn = document.getElementById('loginBtn');
  const logoutBtn = document.getElementById('logoutBtn');

  function updateAuthUI() {
    const user = JSON.parse(localStorage.getItem('worldfestUser') || 'null');
    if (user && user.email) {
      loginBtn.style.display = 'none';
      logoutBtn.style.display = 'inline-block';
      logoutBtn.innerText = `LOGOUT (${user.email})`;
      renderBooked();  // показать билеты, если залогинены
    } else {
      loginBtn.style.display = 'inline-block';
      logoutBtn.style.display = 'none';
      document.getElementById('bookedSection').innerHTML = '<div class="empty-state">Please log in to view your tickets.</div>';
      document.getElementById('purchasedSection').innerHTML = '';
      document.getElementById('cancelledSection').innerHTML = '';
    }
  }

  loginBtn.onclick = () => {
    window.location.href = '123.html';  // перенаправить на главную для логина
  };

  logoutBtn.onclick = () => {
    localStorage.removeItem('worldfestUser');
    updateAuthUI();
    alert('You have been logged out.');
  };

  function renderBooked() {
    const user = JSON.parse(localStorage.getItem('worldfestUser') || 'null');
    if (!user || !user.email) {
      document.getElementById('bookedSection').innerHTML = '<div class="empty-state">Please log in to view your tickets.</div>';
      return;
    }

    const data = JSON.parse(localStorage.getItem('worldfestBookings') || '[]');
    const box = document.getElementById('bookedSection');

    if (!data.length) {
      box.innerHTML = '<div class="empty-state">No booked tickets yet.</div>';
      return;
    }

    box.innerHTML = `
      <div class="tickets-grid">
        ${data.map(item => `
          <div class="ticket-card">
            <h3>${item.concertName || 'Concert'}</h3>
            <div class="ticket-meta"><b>Date:</b> ${item.date || '-'}</div>
            <div class="ticket-meta"><b>Ticket type:</b> ${item.ticketType || '-'}</div>
            <div class="ticket-meta"><b>Name:</b> ${item.name || '-'}</div>
            <div class="ticket-meta"><b>Phone:</b> ${item.phone || '-'}</div>
            <div class="ticket-meta"><b>Price:</b> $${item.price || '-'}</div>
          </div>
        `).join('')}
      </div>
    `;
  }

  function showTab(tab) {
    document.getElementById('bookedSection').style.display = tab === 'booked' ? 'block' : 'none';
    document.getElementById('purchasedSection').style.display = tab === 'purchased' ? 'block' : 'none';
    document.getElementById('cancelledSection').style.display = tab === 'cancelled' ? 'block' : 'none';

    if (tab === 'purchased') {
      document.getElementById('purchasedSection').innerHTML = '<div class="empty-state">No purchased tickets yet.</div>';
    }
    if (tab === 'cancelled') {
      document.getElementById('cancelledSection').innerHTML = '<div class="empty-state">No cancelled tickets yet.</div>';
    }
    if (tab === 'booked') renderBooked();
  }

  document.querySelectorAll('.filter-btn').forEach(btn => {
    btn.addEventListener('click', () => {
      document.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active'));
      btn.classList.add('active');
      showTab(btn.dataset.tab);
    });
  });

  updateAuthUI();
  showTab('booked');
</script>
